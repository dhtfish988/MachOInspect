cmake_minimum_required(VERSION 3.24)
project(MachOInspect VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MACHO_INSPECT_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(MACHO_INSPECT_FUZZ "Build the libFuzzer parser driver" OFF)
find_package(nlohmann_json 3.12 REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0>=2.7)
pkg_check_modules(PCRE2 REQUIRED IMPORTED_TARGET libpcre2-8>=10.40)
add_library(macho_inspect
  src/io/input_source.cpp src/binary/image_catalog.cpp
  src/signing/signing_envelope.cpp src/claims/entitlement_document.cpp
  src/rules/inspection_rules.cpp src/bundle/resource_manifest.cpp
  src/app/inspection_session.cpp src/output/emitters.cpp)
target_include_directories(macho_inspect PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(macho_inspect PUBLIC cxx_std_20)
target_link_libraries(macho_inspect PUBLIC nlohmann_json::nlohmann_json PRIVATE OpenSSL::Crypto PkgConfig::PLIST PkgConfig::PCRE2)
target_compile_options(macho_inspect PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(MACHO_INSPECT_SANITIZE)
  target_compile_options(macho_inspect PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(macho_inspect PUBLIC -fsanitize=address,undefined)
endif()
add_executable(macho-inspect app/command_line.cpp)
target_link_libraries(macho-inspect PRIVATE macho_inspect)
add_executable(inspect-verify-system tools/verify_system.cpp)
target_link_libraries(inspect-verify-system PRIVATE macho_inspect)
include(CTest)
if(BUILD_TESTING)
  add_executable(inspect-tests tests/unit/core_tests.cpp)
  target_link_libraries(inspect-tests PRIVATE macho_inspect PkgConfig::PLIST)
  add_test(NAME core-contract COMMAND inspect-tests)
  add_executable(inspect-resource-tests tests/unit/resource_tests.cpp)
  target_link_libraries(inspect-resource-tests PRIVATE macho_inspect PkgConfig::PLIST)
  add_test(NAME resource-contract COMMAND inspect-resource-tests)
  add_executable(inspect-rule-tests tests/unit/rule_tests.cpp)
  target_link_libraries(inspect-rule-tests PRIVATE macho_inspect)
  add_test(NAME rule-contract COMMAND inspect-rule-tests)
  add_executable(inspect-boundary-tests tests/unit/boundary_tests.cpp)
  target_link_libraries(inspect-boundary-tests PRIVATE macho_inspect)
  add_test(NAME boundary-contract COMMAND inspect-boundary-tests)
  add_executable(inspect-integration tests/integration/macos_contract.cpp)
  target_link_libraries(inspect-integration PRIVATE macho_inspect)
  add_test(NAME macos-integration COMMAND inspect-integration $<TARGET_FILE:macho-inspect>)
  set_tests_properties(macos-integration PROPERTIES SKIP_RETURN_CODE 77)
endif()
if(MACHO_INSPECT_FUZZ)
  target_compile_options(macho_inspect PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(inspect-fuzz tests/fuzz/parser_driver.cpp)
  target_compile_options(inspect-fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(inspect-fuzz PRIVATE -fsanitize=fuzzer)
  target_link_libraries(inspect-fuzz PRIVATE macho_inspect)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/MachOInspectConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/MachOInspectConfig.cmake
  INSTALL_DESTINATION lib/cmake/MachOInspect)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MachOInspectConfigVersion.cmake
  COMPATIBILITY SameMajorVersion)
install(TARGETS macho-inspect macho_inspect EXPORT MachOInspectTargets RUNTIME DESTINATION bin ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT MachOInspectTargets NAMESPACE MachOInspect:: DESTINATION lib/cmake/MachOInspect)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MachOInspectConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/MachOInspectConfigVersion.cmake DESTINATION lib/cmake/MachOInspect)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md dependencies.lock.json
  DESTINATION share/doc/MachOInspect)
install(DIRECTORY docs licenses validation examples DESTINATION share/doc/MachOInspect)
